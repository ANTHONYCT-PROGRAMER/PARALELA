Ejemplo 2: Paralelismo real en 4 núcleos para tareas CPU Bound sin la restricción del GIL

ejemplo2_multiprocessing.py 
 1from multiprocessing import Pool
 2import time
 3
 4def tarea_pesada(n):
 5    # Tarea CPU Bound intensiva
 6    return sum(i * i for i in range(n))
 7
 8if __name__ == "__main__":
 9    datos = [10**7, 10**7, 10**7, 10**7]
10    # Secuencial vs Pool(4)
11    t0 = time.time()
12    res_seq = [tarea_pesada(n) for n in datos]
13    t_seq = time.time() - t0
14    
15    t1 = time.time()
16    with Pool(processes=4) as pool:
17        res_par = pool.map(tarea_pesada, datos)
18    t_par = time.time() - t1
19    print(f"Speedup: {t_seq/t_par:.2f}x")
